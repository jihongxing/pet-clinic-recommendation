```typescript
// logger.config.ts
import { WinstonModule } from 'nest-winston';

import 'winston-daily-rotate-file';

export const loggerConfig = WinstonModule.createLogger({
  transports: [
    // 控制台输出
    new winston.transports.Console({
      format: winston.format.combine(
        winston.format.timestamp(),
        winston.format.colorize(),
        winston.format.printf(({ timestamp, level, message, context, trace }) => {
          return `${timestamp} [${context}] ${level}: ${message}${trace ? `\n${trace}` : ''}`;

      datePattern: 'YYYY-MM-DD',
      level: 'error',
      maxFiles: '30d',
      maxSize: '20m',
      format: winston.format.combine(
        winston.format.timestamp(),
        winston.format.json(),
      ),
    }),
    
    // 所有日志文件
    new winston.transports.DailyRotateFile({
      filename: 'logs/combined-%DATE%.log',
      datePattern: 'YYYY-MM-DD',
      maxFiles: '7d',
      maxSize: '50m',
      format: winston.format.combine(
        winston.format.timestamp(),

  ],
});
```

```typescript
// metrics.service.ts
import { Injectable } from '@nestjs/common';
import { Counter, Histogram, Gauge, register } from 'prom-client';

@Injectable()
export class MetricsService {
  // API请求计数器
  private readonly apiRequestCounter = new Counter({
    name: 'api_requests_total',
    help: 'Total number of API requests',
    labelNames: ['method', 'endpoint', 'status'],
  });
  
  // API响应时间直方图
  private readonly apiResponseTime = new Histogram({
    name: 'api_response_time_seconds',
    help: 'API response time in seconds',
    labelNames: ['method', 'endpoint'],
    buckets: [0.1, 0.5, 1, 2, 5],
  });
  
  // 数据库查询时间
  private readonly dbQueryTime = new Histogram({
    name: 'db_query_duration_seconds',
    help: 'Database query duration in seconds',
    labelNames: ['operation'],
    buckets: [0.01, 0.05, 0.1, 0.5, 1],
  });
  
  // Redis命中率
  private readonly redisCacheHitRate = new Gauge({
    name: 'redis_cache_hit_rate',
    help: 'Redis cache hit rate',
  });
  
  // 标签提交计数
  private readonly tagSubmissionCounter = new Counter({
    name: 'tag_submissions_total',
    help: 'Total number of tag submissions',
    labelNames: ['source', 'layer'],
  });
  
  // 异常检测计数
  private readonly anomalyDetectionCounter = new Counter({
    name: 'anomaly_detections_total',
    help: 'Total number of anomaly detections',
    labelNames: ['type'],
  });
  
  // 记录API请求
  recordApiRequest(method: string, endpoint: string, status: number) {
    this.apiRequestCounter.inc({ method, endpoint, status });
  }
  
  // 记录API响应时间
  recordApiResponseTime(method: string, endpoint: string, duration: number) {
    this.apiResponseTime.observe({ method, endpoint }, duration);
  }
  
  // 记录数据库查询时间
  recordDbQueryTime(operation: string, duration: number) {
    this.dbQueryTime.observe({ operation }, duration);
  }
  
  // 更新Redis命中率
  updateRedisCacheHitRate(hitRate: number) {
    this.redisCacheHitRate.set(hitRate);
  }
  
  // 记录标签提交
  recordTagSubmission(source: string, layer: string) {
    this.tagSubmissionCounter.inc({ source, layer });
  }
  
  // 记录异常检测
  recordAnomalyDetection(type: string) {
    this.anomalyDetectionCounter.inc({ type });
  }
  
  // 暴露指标端点
  getMetrics() {
    return register.metrics();
  }
}
```

#### 监控拦截器

```typescript
// metrics.interceptor.ts
import { Injectable, NestInterceptor, ExecutionContext, CallHandler } from '@nestjs/common';
import { Observable } from 'rxjs';
import { tap } from 'rxjs/operators';
import { MetricsService } from './metrics.service';

@Injectable()
export class MetricsInterceptor implements NestInterceptor {
  constructor(private readonly metricsService: MetricsService) {}
  
  intercept(context: ExecutionContext, next: CallHandler): Observable<any> {
    const request = context.switchToHttp().getRequest();
    const { method, url } = request;
    const startTime = Date.now();
    
    return next.handle().pipe(
      tap({
        next: () => {
          const duration = (Date.now() - startTime) / 1000;
          this.metricsService.recordApiRequest(method, url, 200);
          this.metricsService.recordApiResponseTime(method, url, duration);
        },
        error: (error) => {
          const duration = (Date.now() - startTime) / 1000;
          const status = error.status || 500;
          this.metricsService.recordApiRequest(method, url, status);
          this.metricsService.recordApiResponseTime(method, url, duration);
        },
      }),
    );
  }
}
```

### 2.1 就诊验证系统

#### 功能流程

```
用户点击预约 → 创建order记录（status: clicked）
         ↓
    24小时后推送提醒
         ↓
用户确认"已就诊" → 更新order（status: confirmed）
         ↓
    弹出标签提交页面
         ↓
用户提交标签 → 标签权重1.0（已验证就诊）
```

#### 实现代码

```typescript
// order.service.ts
@Injectable()
export class OrderService {
  constructor(
    private readonly orderRepo: Repository<Order>,
    private readonly queueService: QueueService,
    private readonly notificationService: NotificationService,
  ) {}
  
  // 创建预约记录
  async createOrder(userId: number, clinicId: number, contactType: string) {
    const order = await this.orderRepo.save({
      userId,
      clinicId,
      status: 'clicked',
      contactType,
      createdAt: new Date(),
    });
    
    // 24小时后发送就诊确认提醒
    await this.queueService.addDelayedJob('visit-confirmation', {
      orderId: order.id,
      userId,
      clinicId,
    }, 24 * 60 * 60 * 1000); // 24小时延迟
    
    return order;
  }
  
  // 处理就诊确认提醒
  async sendVisitConfirmation(orderId: number, userId: number, clinicId: number) {
    const order = await this.orderRepo.findOne({ where: { id: orderId } });
    
    if (order && order.status === 'clicked') {
      // 发送小程序订阅消息
      await this.notificationService.sendSubscribeMessage(userId, {
        template_id: 'visit_confirmation_template',
        page: `/pages/visit-confirm/index?orderId=${orderId}`,
        data: {
          thing1: { value: '就诊确认' },
          thing2: { value: '请确认您是否已就诊' },
          time3: { value: order.createdAt.toLocaleString() },
        },
      });
    }
  }
  
  // 用户确认已就诊
  async confirmVisit(userId: number, orderId: number, visited: boolean) {
    const order = await this.orderRepo.findOne({ 
      where: { id: orderId, userId } 
    });
    
    if (!order) {
      throw new Error('订单不存在');
    }
    
    if (visited) {
      // 确认已就诊
      await this.orderRepo.update(orderId, {
        status: 'confirmed',
        confirmedAt: new Date(),
      });
      
      // 返回标志，前端跳转到标签提交页面
      return { shouldSubmitTag: true, clinicId: order.clinicId };
    } else {
      // 未就诊
      await this.orderRepo.update(orderId, {
        status: 'cancelled',
      });
      
      return { shouldSubmitTag: false };
    }
  }
}
```

### 2.2 诊所回应功能

#### 功能设计

**场景**：诊所对负面标签进行回应

**实现方式**：
1. 诊所认领后可以对标签进行回应
2. 回应内容显示在标签下方
3. 回应需要审核（防止辱骂用户）

#### 数据表设计

```sql
CREATE TABLE clinic_tag_response (
  id SERIAL PRIMARY KEY,
  clinic_id INT NOT NULL,
  tag_id INT NOT NULL,
  response_text TEXT NOT NULL,
  status VARCHAR(20) DEFAULT 'pending', -- pending/approved/rejected
  created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
  approved_at TIMESTAMP,
  approved_by INT,
  FOREIGN KEY (clinic_id) REFERENCES clinic(id),
  FOREIGN KEY (tag_id) REFERENCES tag(id),
  UNIQUE KEY uk_clinic_tag (clinic_id, tag_id)
);
```

#### 实现代码

```typescript
// clinic-response.service.ts
@Injectable()
export class ClinicResponseService {
  constructor(
    private readonly responseRepo: Repository<ClinicTagResponse>,
    private readonly clinicRepo: Repository<Clinic>,
  ) {}
  
  // 诊所提交回应
  async submitResponse(clinicId: number, tagId: number, responseText: string) {
    // 检查诊所是否已认领
    const clinic = await this.clinicRepo.findOne({ where: { id: clinicId } });
    if (!clinic.isClaimed) {
      throw new Error('请先认领诊所');
    }
    
    // 检查是否已有回应
    const existing = await this.responseRepo.findOne({
      where: { clinicId, tagId },
    });
    
    if (existing) {
      // 更新回应
      await this.responseRepo.update(existing.id, {
        responseText,
        status: 'pending',
        createdAt: new Date(),
      });
    } else {
      // 创建新回应
      await this.responseRepo.save({
        clinicId,
        tagId,
        responseText,
        status: 'pending',
      });
    }
    
    return { success: true };
  }
  
  // 管理员审核回应
  async approveResponse(responseId: number, adminId: number, approved: boolean) {
    await this.responseRepo.update(responseId, {
      status: approved ? 'approved' : 'rejected',
      approvedAt: new Date(),
      approvedBy: adminId,
    });
  }
  
  // 获取诊所的标签回应
  async getClinicResponses(clinicId: number) {
    return await this.responseRepo.find({
      where: { clinicId, status: 'approved' },
      relations: ['tag'],
    });
  }
}
```

### 2.3 数据分析管道

#### 分析维度

**1. 用户行为分析**
- 转化漏斗：浏览 → 查看详情 → 点击预约 → 确认就诊 → 提交标签
- 标签提交率
- 用户留存率

**2. 诊所数据分析**
- 标签增长趋势
- 预约转化率
- 风险标签预警

**3. 反作弊分析**
- 异常提交模式检测
- 可疑用户识别

#### 实现方案

```typescript
// analytics.service.ts
@Injectable()
export class AnalyticsService {
  constructor(
    private readonly db: DataSource,
    private readonly redis: Redis,
  ) {}
  
  // 计算转化漏斗
  async calculateConversionFunnel(startDate: Date, endDate: Date) {
    const result = await this.db.query(`
      SELECT 
        COUNT(DISTINCT u.id) as total_users,
        COUNT(DISTINCT CASE WHEN o.id IS NOT NULL THEN u.id END) as clicked_users,
        COUNT(DISTINCT CASE WHEN o.status = 'confirmed' THEN u.id END) as confirmed_users,
        COUNT(DISTINCT CASE WHEN t.id IS NOT NULL THEN u.id END) as tagged_users
      FROM user u
      LEFT JOIN \`order\` o ON u.id = o.user_id 
        AND o.created_at BETWEEN ? AND ?
      LEFT JOIN user_tag_log t ON u.id = t.user_id 
        AND t.created_at BETWEEN ? AND ?
      WHERE u.created_at BETWEEN ? AND ?
    `, [startDate, endDate, startDate, endDate, startDate, endDate]);
    
    const data = result[0];
    return {
      totalUsers: data.total_users,
      clickRate: data.clicked_users / data.total_users,
      confirmRate: data.confirmed_users / data.clicked_users,
      tagSubmitRate: data.tagged_users / data.confirmed_users,
    };
  }
  
  // 诊所标签增长趋势
  async getClinicTagGrowth(clinicId: number, days: number = 30) {
    const result = await this.db.query(`
      SELECT 
        DATE(created_at) as date,
        COUNT(*) as tag_count,
        COUNT(DISTINCT user_id) as unique_users
      FROM user_tag_log
      WHERE clinic_id = ?
        AND created_at >= DATE_SUB(NOW(), INTERVAL ? DAY)
      GROUP BY DATE(created_at)
      ORDER BY date
    `, [clinicId, days]);
    
    return result;
  }
  
  // 检测异常提交模式
  async detectAbnormalPatterns() {
    // 1. 短时间内大量相同标签
    const sameTagSpam = await this.db.query(`
      SELECT 
        clinic_id,
        tag_id,
        COUNT(*) as count,
        COUNT(DISTINCT user_id) as unique_users
      FROM user_tag_log
      WHERE created_at > DATE_SUB(NOW(), INTERVAL 1 HOUR)
      GROUP BY clinic_id, tag_id
      HAVING count > 10 AND unique_users < 3
    `);
    
    // 2. 同一用户短时间内标记多个诊所
    const userSpam = await this.db.query(`
      SELECT 
        user_id,
        COUNT(DISTINCT clinic_id) as clinic_count
      FROM user_tag_log
      WHERE created_at > DATE_SUB(NOW(), INTERVAL 1 HOUR)
      GROUP BY user_id
      HAVING clinic_count > 5
    `);
    
    return {
      sameTagSpam,
      userSpam,
    };
  }
  
  // 生成每日报表
  @Cron('0 1 * * *') // 每天凌晨1点
  async generateDailyReport() {
    const yesterday = new Date();
    yesterday.setDate(yesterday.getDate() - 1);
    yesterday.setHours(0, 0, 0, 0);
    
    const today = new Date();
    today.setHours(0, 0, 0, 0);
    
    const funnel = await this.calculateConversionFunnel(yesterday, today);
    const anomalies = await this.detectAbnormalPatterns();
    
    // 发送报表到管理员
    await this.sendDailyReport({
      date: yesterday.toISOString().split('T')[0],
      funnel,
      anomalies,
    });
  }
}
```
